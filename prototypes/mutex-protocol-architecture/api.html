为保证原子性，try_acquire 和 release 使用 Lua 脚本实现，避免竞态条件。

-- 只有当 key 不存在或已有 key 的 owner == device_id 时才写入
-- 避免 A 还拿着过期锁时 B 抢锁成功
local key      = KEYS[1]
local owner    = ARGV[1]
local term     = tonumber(ARGV[2])
local value    = ARGV[3]
local ttl      = tonumber(ARGV[4])

local existing = redis.call("GET", key)
if existing then
    local prev = cjson.decode(existing)
    -- 同一个 device 续约？允许；否则拒绝
    if prev["owner"] == owner then
        redis.call("SETEX", key, ttl, value)
        return 1
    else
        return 0
    end
else
    redis.call("SETEX", key, ttl, value)
    return 1
end

LockManager 类

python · backend/plugins/mutex_protocol/lock.py 
import json
import logging
import time
from typing import Optional, Dict

log = logging.getLogger(__name__)

LOCK_PREFIX = "mutex:lock:"
LOCK_TTL_SEC = 30


class LockManager:
    """基于 Redis SET NX 的分布式互斥锁。"""

    def __init__(self, redis_client):
        self._redis = redis_client

    def _key(self, group_wxid: str) -> str:
        return f"{LOCK_PREFIX}{group_wxid}"

    def try_acquire(
        self, group_wxid: str, device_id: str, term: int
    ) -> bool:
        """尝试获取锁（SET NX + EX 30）。

        term: 任期号，用于 fencing token —— 只有新任期能覆盖旧任期。
        """
        key = self._key(group_wxid)
        value = json.dumps({
            "owner": device_id,
            "term": term,
            "lease": int(time.time()) + LOCK_TTL_SEC,
            "ts": int(time.time()),
        })
        # SET NX EX 原子操作
        ok = self._redis.set(key, value, nx=True, ex=LOCK_TTL_SEC)
        if ok:
            log.info("🔒 抢锁成功 group=%s owner=%s term=%d", group_wxid[:20], device_id, term)
        else:
            log.debug("🔓 抢锁失败 group=%s owner=%s term=%d", group_wxid[:20], device_id, term)
        return bool(ok)

    def renew(self, group_wxid: str, device_id: str, term: int) -> bool:
        """Leader 续租：只有 owner 和 term 都匹配才允许延长 TTL。"""
        key = self._key(group_wxid)
        raw = self._redis.get(key)
        if not raw:
            return False
        try:
            data = json.loads(raw)
        except json.JSONDecodeError:
            return False

        if data["owner"] != device_id or data["term"] != term:
            return False  # 锁已易主

        # 续租：重写 value + 延长 TTL
        data["lease"] = int(time.time()) + LOCK_TTL_SEC
        data["ts"] = int(time.time())
        self._redis.setex(key, LOCK_TTL_SEC, json.dumps(data))
        return True

    def release(self, group_wxid: str, device_id: str, term: int) -> bool:
        """主动释放锁（Lua 脚本校验 owner，避免删了别人的锁）。"""
        key = self._key(group_wxid)
        raw = self._redis.get(key)
        if not raw:
            return True
        try:
            data = json.loads(raw)
        except json.JSONDecodeError:
            return True

        if data["owner"] != device_id or data["term"] != term:
            log.warning("⚠️ 拒绝释放：lock 已易主 group=%s local=%d remote=%d",
                        group_wxid, term, data.get("term"))
            return False

        self._redis.delete(key)
        return True

    def peek(self, group_wxid: str) -> Optional[Dict]:
        """读锁内容（不修改）。"""
        key = self._key(group_wxid)
        raw = self._redis.get(key)
        if not raw:
            return None
        try:
            return json.loads(raw)
        except json.JSONDecodeError:
            return None

心跳续租与本地状态

python · backend/plugins/mutex_protocol/monitor.py 
"""DeviceMonitor: Leader 心跳续租 + 本地状态管理。

设计要点：
1. 每 10 秒尝试续租一次 → 3 个周期内都失败就降级
2. 线程安全：本地 dict 用 threading.Lock 保护
3. 所有异常只记日志，绝不影响消息处理主链路
"""

import logging
import threading
import time

log = logging.getLogger(__name__)

class DeviceMonitor:
    HEARTBEAT_INTERVAL = 10   # 心跳周期（秒）
    MAX_FAILURES = 3           # 连续失败上限 → 降级

    def __init__(self, lock_manager, device_id: str):
        self._lock_mgr = lock_manager
        self._device_id = device_id
        self._state_lock = threading.Lock()

        # 本地状态：{group_wxid: {leader, term, failures, timer, running}}
        self._groups: dict = {}

    def promote(self, group_wxid: str, term: int):
        """当选 Leader 后调用：启动心跳续租线程。"""
        with self._state_lock:
            state = self._groups.setdefault(group_wxid, {
                "leader": None,
                "term": 0,
                "failures": 0,
                "timer": None,
            })
            state["leader"] = self._device_id
            state["term"] = term
            state["failures"] = 0
            if not state["timer"]:
                state["timer"] = threading.Thread(
                    target=self._heartbeat_loop, args=(group_wxid,), daemon=True,
                )
                state["timer"].start()

    def demote(self, group_wxid: str, reason: str = ""):
        """降级为 Follower：停止心跳、清空 leader 标记。"""
        with self._state_lock:
            state = self._groups.get(group_wxid)
            if state:
                state["leader"] = None
                state["failures"] = 0
        log.info("""⚙️  降级 Follower group=%s reason=%s""", group_wxid[:20], reason)

    def is_leader(self, group_wxid: str) -> bool:
        with self._state_lock:
            state = self._groups.get(group_wxid)
            return bool(state and state["leader"] == self._device_id)

    def heartbeat_ok(self, group_wxid: str) -> bool:
        with self._state_lock:
            state = self._groups.get(group_wxid)
            return bool(state and state["failures"] < self.MAX_FAILURES)

    def _heartbeat_loop(self, group_wxid):
        while True:
            time.sleep(self.HEARTBEAT_INTERVAL)
            with self._state_lock:
                state = self._groups.get(group_wxid)
                if not state or state["leader"] != self._device_id:
                    break  # 已降级，退出心跳线程
                term = state["term"]

            try:
                ok = self._lock_mgr.renew(group_wxid, self._device_id, term)
            except Exception as exc:
                log.error("心跳异常 group=%s: %s", group_wxid[:20], exc)
                ok = False

            with self._state_lock:
                state = self._groups.get(group_wxid)
                if not state:
                    break
                if ok:
                    state["failures"] = 0
                else:
                    state["failures"] += 1
                    log.warning("心跳失败 %d/%d group=%s",
                                state["failures"], self.MAX_FAILURES, group_wxid[:20])
                    if state["failures"] >= self.MAX_FAILURES:
                        log.error("""🚨 心跳超时，主动降级 group=%s""", group_wxid[:20])
                        state["leader"] = None
                        break

插件入口模块

插件对外暴露的唯一公共 API，供 message_sync.py 调用。

python · backend/plugins/mutex_protocol/__init__.py 
"""互斥控制插件 · 公共 API 入口。

用法（在 message_sync.py 的 broadcast_messages 中）：
    from backend.plugins.mutex_protocol import should_process
    if not should_process(conversation_id):
        continue
    on_new_message(conversation_id, ...)
"""

import logging
from typing import Optional, Dict

log = logging.getLogger(__name__)

# 单例实例（由 init_app 初始化）
_lock_manager  = None
_device_monitor = None
_local_terms: dict = {}  # {group_wxid: int}
_local_lock: threading.Lock = threading.Lock()


def init_app(app, redis_client, device_id: str):
    """应用启动时调用：初始化锁管理器和心跳监控。"""
    from .lock import LockManager
    from .monitor import DeviceMonitor
    import threading  # noqa

    global _lock_manager, _device_monitor
    _lock_manager = LockManager(redis_client)
    _device_monitor = DeviceMonitor(_lock_manager, device_id)
    log.info("🔐 MutexProtocol 已初始化 device_id=%s", device_id)


def should_process(conversation_id: int) -> bool:
    """插件主入口：判定当前设备是否该处理该会话的消息。

    返回 True  → 放行到业务逻辑（自动报价执行）
    返回 False → 静默丢弃（Follower 设备）
    """
    if not _lock_manager or not _device_monitor:
        log.warning("MutexProtocol 未初始化，默认放行")
        return True

    with _local_lock:
        try:
            from backend.models.conversation import Conversation
            from backend.extensions import db

            conv = db.session.get(Conversation, conversation_id)
            if not conv or conv.chat_type != Conversation.CHAT_GROUP:
                return True  # 非群聊，无需互斥

            group_wxid = conv.wxid

            # 快速路径：本地已知自己是 Leader 且心跳正常
            if _device_monitor.is_leader(group_wxid) and _device_monitor.heartbeat_ok(group_wxid):
                return True

            # 查 Redis 锁
            lock = _lock_manager.peek(group_wxid)

            if lock is not None:
                if lock["owner"] == _device_monitor._device_id:
                    return True  # 兜底：锁还是自己的

                # 别人有锁，看任期号
                local_term = _local_terms.get(group_wxid, 0)
                if lock["term"] > local_term:
                    _local_terms[group_wxid] = lock["term"]
                return False  # ❌ 静默

            # 锁空 → 选举：抢锁
            local_term = _local_terms.get(group_wxid, 0)
            new_term = local_term + 1

            if _lock_manager.try_acquire(group_wxid, _device_monitor._device_id, new_term):
                _local_terms[group_wxid] = new_term
                _device_monitor.promote(group_wxid, new_term)
                return True  # 🎉 当选

            # 抢锁失败 → 读 winner 同步状态
            winner = _lock_manager.peek(group_wxid)
            if winner:
                _local_terms[group_wxid] = winner["term"]
            return False

        except Exception as exc:
            # 任何异常都兜底放行 —— 插件不能阻断业务
            log.exception("MutexProtocol should_process 异常: %s", exc)
            return True
