def should_process(conversation_id: int) -> bool:
    """插件拦截入口：返回 True = 放行到业务逻辑；False = 静默丢弃。"""

    conv = Conversation.query.get(conversation_id)
    if not conv:
        return True  # 会话不存在，让下游自己处理

    # 只在群聊场景生效（单聊天然只有一个设备，不需要互斥）
    if conv.chat_type != Conversation.CHAT_GROUP:
        return True

    group_wxid = conv.wxid
    device_id  = current_device.openid
    term       = local_term.get(group_wxid, 0)

    # --- 快速路径：本地已知自己是 Leader，且心跳正常 ---
    if local_leader.get(group_wxid) == device_id:
        if heartbeat_ok(group_wxid):
            return True     # ✅ 放行，继续报价
        else:
            # 心跳异常 → 可能脑裂 → 降级为被动，等选举
            local_leader[group_wxid] = None

    # --- 查 Redis 锁 ---
    lock = lock_manager.peek(group_wxid)

    if lock is not None:
        if lock["owner"] == device_id:
            # 锁还是自己的（快速路径已覆盖，这里兜底）
            return True
        if lock["term"] > term:
            # ✅ 别人任期更高 → 静默
            local_term[group_wxid] = lock["term"]
            return False
        # term 相等或更低但 owner 不是自己 → 锁异常，尝试抢

    # --- 选举：尝试抢锁 ---
    new_term = max(term, (lock and lock["term"])) + 1
    if lock_manager.try_acquire(group_wxid, device_id, new_term):
        local_leader[group_wxid] = device_id
        local_term[group_wxid]   = new_term
        start_heartbeat(group_wxid)        # 启动心跳续租
        return True     # 🎉 当选 Leader，放行
    else:
        # 抢锁失败 → 别人当选了，读它的 term 同步本地状态
        winner = lock_manager.peek(group_wxid)
        if winner:
            local_term[group_wxid] = winner["term"]
        return False  # ❌ 静默丢弃
